# Quasar v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A local Bun service that generates and incrementally edits Next.js 15 sites with Claude (Tool Runner + path-guarded file tools) plus an Expo app that drives it, with local `next dev` previews.

**Architecture:** Bun-workspaces monorepo. `packages/service` is a `Bun.serve` HTTP API owning a file-based site registry, a scaffolder that copies `templates/next-site`, a generation engine built on `@anthropic-ai/sdk`'s beta Tool Runner (three zod tools scoped to one site directory), and a preview manager that spawns `next dev` per site. `apps/quasar` is an Expo Router app that lists sites, streams generation events (JSON cursor polling), and controls previews. `packages/shared` holds the API contract.

**Tech Stack:** Bun (runtime, workspaces, `bun:test`), TypeScript strict, `@anthropic-ai/sdk` + `zod`, Next.js 15 App Router + Tailwind v4 (generated sites), Expo SDK 53 / React 19 / RN 0.79 / Expo Router 5 (app).

**Spec:** `docs/superpowers/specs/2026-08-19-quasar-v1-design.md`

## Global Constraints

- Bun only — never npm/yarn/pnpm. Repo root: `/Volumes/Extreme SSD/quasar` (path has a space — always quote it in shell commands).
- TypeScript `strict: true` everywhere; every package has a `typecheck` script (`tsc --noEmit`).
- Model: `claude-opus-5`, `thinking: {type: "adaptive"}`, `max_tokens: 64000`. Credentials resolve via the SDK's default chain — never store/proxy a key.
- Path guard: every model-supplied path must resolve inside the site dir; reject absolute paths, `..`, symlink escapes, `.git`/`node_modules`/`.next` segments, writes > 512 KB.
- Service data root: `process.env.QUASAR_HOME ?? path.join(os.homedir(), ".quasar")` — tests always set `QUASAR_HOME` to a temp dir.
- Tests: `bun:test` (`import { test, expect } from "bun:test"`). No network in tests; the Anthropic client is always injected.
- Commit after each task with a conventional message; the required commit trailer lines are in the repo session rules.

**Worker note — versions:** dependency versions below use `latest`/caret pins where the exact current version is unknown; `bun install` / `bunx expo install --fix` resolves them. Never invent an exact version pin not shown here.

```ts
import path from "node:path";
import { realpath } from "node:fs/promises";

export class PathGuardError extends Error {}
export const MAX_FILE_BYTES = 512 * 1024;
const BLOCKED = new Set([".git", "node_modules", ".next"]);

export async function resolveSitePath(siteDir: string, relPath: string): Promise<string> {
  if (path.isAbsolute(relPath)) throw new PathGuardError(`absolute path not allowed: ${relPath}`);
  const segments = relPath.split(/[\\/]+/).filter(Boolean);
  if (segments.some((s) => s === "..")) throw new PathGuardError(`path traversal not allowed: ${relPath}`);
  if (segments.some((s) => BLOCKED.has(s))) throw new PathGuardError(`blocked path segment in: ${relPath}`);
  const rootReal = await realpath(siteDir);
  const abs = path.join(rootReal, ...segments);
  // realpath the deepest existing ancestor to catch symlink escapes
  let probe = abs;
  while (true) {
    try {
      const real = await realpath(probe);
      if (real !== rootReal && !real.startsWith(rootReal + path.sep)) {
        throw new PathGuardError(`path escapes site dir: ${relPath}`);
      }
      break;
    } catch (err) {
      if (err instanceof PathGuardError) throw err;
      const parent = path.dirname(probe);
      if (parent === probe) throw new PathGuardError(`unresolvable path: ${relPath}`);
      probe = parent;
    }
  }
  return path.join(rootReal, ...segments);
}
```

```ts
import { betaZodTool } from "@anthropic-ai/sdk/helpers/beta/zod";
import { z } from "zod";
import { listSiteFiles, readSiteFile, writeSiteFile } from "./siteFs";
import type { GenerationEvent } from "@quasar/shared";
import { buildSystemPrompt } from "./systemPrompt";

export function createSiteTools(siteDir: string, onEvent: (ev: GenerationEvent) => void) {
  return [
    betaZodTool({
      name: "list_files",
      description: "List every file in the site as relative paths.",
      inputSchema: z.object({}),
      run: async () => { onEvent({ type: "tool", name: "list_files" }); return (await listSiteFiles(siteDir)).join("\n"); },
    }),
    betaZodTool({
      name: "read_file",
      description: "Read one file from the site by relative path.",
      inputSchema: z.object({ path: z.string() }),
      run: async (input) => { onEvent({ type: "tool", name: "read_file", path: input.path }); return readSiteFile(siteDir, input.path); },
    }),
    betaZodTool({
      name: "write_file",
      description: "Create or overwrite one file in the site. Always write complete file contents.",
      inputSchema: z.object({ path: z.string(), content: z.string() }),
      run: async (input) => {
        await writeSiteFile(siteDir, input.path, input.content);
        onEvent({ type: "tool", name: "write_file", path: input.path });
        return `wrote ${input.path}`;
      },
    }),
  ];
}

export async function runGeneration({ client, siteDir, prompt, onEvent }: { client: EngineClient; siteDir: string; prompt: string; onEvent: (ev: GenerationEvent) => void }): Promise<void> {
  try {
    const runner = client.beta.messages.toolRunner({
      model: "claude-opus-5",
      max_tokens: 64000,
      thinking: { type: "adaptive" },
      system: buildSystemPrompt(),
      tools: createSiteTools(siteDir, onEvent),
      messages: [{ role: "user", content: prompt }],
    });
    for await (const message of runner) {
      for (const block of message.content as Array<{ type: string; text?: string }>) {
        if (block.type === "text" && block.text) onEvent({ type: "text", text: block.text });
      }
    }
    onEvent({ type: "done" });
  } catch (err) {
    onEvent({ type: "error", message: err instanceof Error ? err.message : String(err) });
  }
}
```
